## 构建 openai
```shell
#安装插件
go install github.com/google/gnostic/cmd/protoc-gen-openapi@latest

# 生成 yaml 文件
protoc --proto_path=. --openapi_out=fq_schema_naming=true,default_response=false:. lesson.proto

linux 系统需要设置 GOPATH
go env -w GO111MODULE=on
```

## wire 依赖注入
> 是 go 语言的以来注入工具
> 通过代码生成来自动化组件间的连接过程

```shell
#1. 下载插件
go install github.com/google/wire/cmd/wire@latest
```
```go
// 1. 创建构造代码
package main

type Config struct {
	RedisIp string
	RedisPort string
	MysqlIp string
	MysqlPort string
}

type MysqlClient struct {}
type RedisClient struct {}

type App struct {}

func NewConfig() Config {
	return Config{}
}
func NewMysqlClient(config Config) MysqlClient {
	return MysqlClient{}
}
func NewRedisClient(config Config) RedisClient {
	return RedisClient{}
}

func NewApp(m MysqlClient, r RedisClient) App {
	return App{}
}

//2. 创建依赖注入文件
//go:build wireinject
// +build wireinject

package main

import "github.com/google/wire"

func WireApp() App {
	wire.Build(NewConfig, NewRedisClient, NewMysqlClient, NewApp)
	return App{}
}

//3. 在终端使用命令 wire 生成
// wire

// 4. 生成的代码
// Code generated by Wire. DO NOT EDIT.

//go:generate go run -mod=mod github.com/google/wire/cmd/wire
//go:build !wireinject
// +build !wireinject

package main

// Injectors from wire.go:

func WireApp() App {
	config := NewConfig()
	mysqlClient := NewMysqlClient(config)
	redisClient := NewRedisClient(config)
	app := NewApp(mysqlClient, redisClient)
	return app
}
```
